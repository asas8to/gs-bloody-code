#!/usr/bin/env python3 """ TeacherPy - a tiny language interpreter written in Python. Syntax highlights: - Comments start with # - Assignment: x = 5 - Print: print x, "text", (1+2) - If: if (x > 0) { ... } else { ... } - While: while (x < 10) { ... } - Blocks use { } and can contain multiple statements. - Expressions: numbers, strings "..." , variables, + - * / % == != < <= >= and parentheses. """ import re import sys # ---------- Lexer ---------- TOKEN_SPEC = [ ('NUMBER', r'\d+(\.\d+)?'), ('STRING', r'"([^"\\]|\\.)*"'), ('ID', r'[A-Za-z_][A-Za-z0-9_]*'), ('OP', r'==|!=|<=|>=|[+\-*/%<>=]'), ('LPAREN', r'\('), ('RPAREN', r'\)'), ('LBRACE', r'\{'), ('RBRACE', r'\}'), ('COMMA', r','), ('NEWLINE', r'\n'), ('SKIP', r'[ \t]+'), ('MISMATCH', r'.'), ] TOK_REGEX = re.compile('|'.join('(?P<%s>%s)' % pair for pair in TOKEN_SPEC)) KEYWORDS = {'print', 'if', 'else', 'while'} class Token: def __init__(self, typ, val): self.type = typ self.val = val def __repr__(self): return f"Token({self.type},{self.val})" def lex(text): pos = 0 tokens = [] while pos < len(text): m = TOK_REGEX.match(text, pos) if not m: raise SyntaxError(f"Unexpected char at {pos}") kind = m.lastgroup val = m.group(kind) pos = m.end() if kind == 'NUMBER': tokens.append(Token('NUMBER', float(val) if '.' in val else int(val))) elif kind == 'STRING': # remove quotes and handle escapes s = val[1:-1].encode('utf-8').decode('unicode_escape') tokens.append(Token('STRING', s)) elif kind == 'ID': if val in KEYWORDS: tokens.append(Token(val.upper(), val)) else: tokens.append(Token('ID', val)) elif kind == 'OP': tokens.append(Token('OP', val)) elif kind in ('LPAREN','RPAREN','LBRACE','RBRACE','COMMA'): tokens.append(Token(kind, val)) elif kind == 'NEWLINE': tokens.append(Token('NEWLINE', val)) elif kind == 'SKIP': continue elif kind == 'MISMATCH': if val == '#': # comment: skip till end of line while pos < len(text) and text[pos] != '\n': pos += 1 continue raise SyntaxError(f"Unexpected token: {val}") tokens.append(Token('EOF', None)) return tokens # ---------- Parser (simple statement-level parser) ---------- class Parser: def __init__(self, tokens): self.tokens = tokens self.i = 0 def cur(self): return self.tokens[self.i] def eat(self, ttype=None): tok = self.cur() if ttype and tok.type != ttype: raise SyntaxError(f"Expected {ttype}, got {tok.type}") self.i += 1 return tok # Pratt-like expression parser using operator precedence (shunting-yard approach inside) def parse_expression(self): # convert token stream for expression into RPN using shunting-yard outq = [] opstack = [] def precedence(op): if op in ('==','!=','<','>','<=','>='): return 2 if op in ('+','-'): return 3 if op in ('*','/','%'): return 4 return 0 def push_op(op): while opstack and opstack[-1] != '(' and precedence(opstack[-1]) >= precedence(op): outq.append(('OP', opstack.pop())) opstack.append(op) while True: tok = self.cur() if tok.type == 'NUMBER' or tok.type == 'STRING': outq.append((tok.type, tok.val)); self.i += 1 elif tok.type == 'ID': outq.append(('ID', tok.val)); self.i += 1 elif tok.type == 'LPAREN': opstack.append('('); self.i += 1 elif tok.type == 'RPAREN': while opstack and opstack[-1] != '(': outq.append(('OP', opstack.pop())) if not opstack: raise SyntaxError("Mismatched )") opstack.pop() # pop '(' self.i += 1 elif tok.type == 'OP': push_op(tok.val); self.i += 1 else: break while opstack: op = opstack.pop() if op == '(': raise SyntaxError("Mismatched (") outq.append(('OP', op)) return ('RPN', outq) def parse_statement(self): tok = self.cur() # empty/newline skip if tok.type == 'NEWLINE': self.eat('NEWLINE') return None # print statement if tok.type == 'PRINT': self.eat('PRINT') expr = self.parse_expression() # optional newline if self.cur().type == 'NEWLINE': self.eat('NEWLINE') return ('PRINT', expr) # if statement: if (expr) { block } [else { block }] if tok.type == 'IF': self.eat('IF') # expect LPAREN if self.cur().type == 'LPAREN': self.eat('LPAREN') cond = self.parse_expression() if self.cur().type == 'RPAREN': self.eat('RPAREN') if self.cur().type != 'LBRACE': raise SyntaxError("Expected { after if condition") then_block = self.parse_block() else_block = None if self.cur().type == 'ELSE': self.eat('ELSE') if self.cur().type != 'LBRACE': raise SyntaxError("Expected { after else") else_block = self.parse_block() return ('IF', cond, then_block, else_block) # while statement: while (expr) { block } if tok.type == 'WHILE': self.eat('WHILE') if self.cur().type == 'LPAREN': self.eat('LPAREN') c
